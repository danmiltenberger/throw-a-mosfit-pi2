#include <SparkFun_TB6612.h>

// ---------- Pin definitions ----------
// Front driver
#define F_PWMA 7
#define F_AIN2 34
#define F_AIN1 36
#define F_STBY 38
#define F_BIN1 40
#define F_BIN2 42
#define F_PWMB 6

// Back driver
#define B_PWMA 5
#define B_AIN1 35
#define B_AIN2 33
#define B_STBY 37
#define B_BIN1 39
#define B_BIN2 41
#define B_PWMB 4


// IR
#define left_IR_analog A0
#define left_IR_digital 42

#define right_IR_analog A1
#define right_IR_digital 43

// Direction offsets: use 1 normally, or -1 to flip a motor's direction
const int OFFSET_FL = -1;
const int OFFSET_FR = 1;
const int OFFSET_BL = -1;
const int OFFSET_BR = 1;

// ---------- Settings ----------
const int DRIVE_SPEED = 150;            // 0-255, used for serial commands
const float MAX_DURATION_S = 30.0;      // safety cap on a single command

// ---------- Test settings ----------
const int TEST_SPEED = 150;     // 0-255
const int RUN_TIME_MS = 1500;   // how long each direction runs
const int PAUSE_MS = 500;       // pause between steps

// ---------- Motors ----------
//Motor frontLeft  = Motor(F_AIN1, F_AIN2, F_PWMA, OFFSET_FL, F_STBY);     
//Motor frontRight = Motor(F_BIN1, F_BIN2, F_PWMB, OFFSET_FR, F_STBY);     
//Motor backLeft   = Motor(B_AIN1, B_AIN2, B_PWMA, OFFSET_BL, B_STBY);     
//Motor backRight  = Motor(B_BIN1, B_BIN2, B_PWMB, OFFSET_BR, B_STBY);     

Motor frontLeft   = Motor(B_AIN1, B_AIN2, B_PWMA, OFFSET_FL, B_STBY);
Motor frontRight  = Motor(B_BIN1, B_BIN2, B_PWMB, OFFSET_FR, B_STBY);   // correct

Motor backLeft = Motor(F_BIN1, F_BIN2, F_PWMB, OFFSET_BL, F_STBY);      
Motor backRight  = Motor(F_AIN1, F_AIN2, F_PWMA, OFFSET_BR, F_STBY);    // correct


// ---------- State frontLeft----------
bool moving = false;
unsigned long moveStartMs = 0;
unsigned long moveDurationMs = 0;

String inputBuffer = "";

// ---------- Motion primitives ----------
void setMotors(int fl, int fr, int bl, int br) {
  frontLeft.drive(fl);
  frontRight.drive(fr);
  backLeft.drive(bl);
  backRight.drive(br);
}

void stopAllMotors() {
  frontLeft.brake();
  frontRight.brake();
  backLeft.brake();
  backRight.brake();
}

void forward(int s)        { setMotors( s,  s,  s,  s); }
void backward(int s)       { setMotors(-s, -s, -s, -s); }
void strafeLeft(int s)     { setMotors(-s,  s,  s, -s); }
void strafeRight(int s)    { setMotors( s, -s, -s,  s); }
void rotateCCW(int s)      { setMotors(-s,  s, -s,  s); }  // left side back, right side forward
void rotateCW(int s)       { setMotors( s, -s,  s, -s); }  // left side forward, right side back

// ---------- Startup motor test ----------
void testOneMotor(const char *name, Motor &m) {
  Serial.print(name);
  Serial.println(" forward");
  m.drive(TEST_SPEED);
  delay(RUN_TIME_MS);
  m.brake();
  delay(PAUSE_MS);

  Serial.print(name);
  Serial.println(" backward");
  m.drive(-TEST_SPEED);
  delay(RUN_TIME_MS);
  m.brake();
  delay(PAUSE_MS);
}

void testMotors() {
  testOneMotor("front left",  frontLeft);
  testOneMotor("front right", frontRight);
  testOneMotor("back left",   backLeft);
  testOneMotor("back right",  backRight);
  Serial.println("Motor test complete");
}

void runStartupTest() {
  testMotors();

  Serial.println("all motors forward");
  forward(TEST_SPEED);
  delay(2000);

  Serial.println("all motors backward");
  backward(TEST_SPEED);
  delay(2000);

  stopAllMotors();
}

// ---------- Command handling ----------
// Command format: <letter><seconds>   e.g. W1, S0.5, Q1.2, E2
//   W forward   S backward   A left   D right
//   Q rotate counterclockwise   E rotate clockwise
//   X stop immediately
void handleCommand(String cmd) {
  cmd.trim();
  cmd.toUpperCase();
  if (cmd.length() == 0) return;

  char dir = cmd.charAt(0);

  if (dir == 'X') {
    stopAllMotors();
    moving = false;
    Serial.println("Stopped");
    return;
  }

  if (cmd.length() < 2) {
    Serial.println("Error: missing duration (example: W1.5)");
    return;
  }

  float seconds = cmd.substring(1).toFloat();
  if (seconds <= 0) {
    Serial.println("Error: invalid duration");
    return;
  }
  if (seconds > MAX_DURATION_S) {
    seconds = MAX_DURATION_S;
    Serial.println("Duration capped at max");
  }

  switch (dir) {
    case 'W': forward(DRIVE_SPEED);     Serial.print("Forward ");     break;
    case 'S': backward(DRIVE_SPEED);    Serial.print("Backward ");    break;
    case 'A': strafeLeft(DRIVE_SPEED);  Serial.print("Left ");        break;
    case 'D': strafeRight(DRIVE_SPEED); Serial.print("Right ");       break;
    case 'Q': rotateCCW(DRIVE_SPEED);   Serial.print("Rotate CCW ");  break;
    case 'E': rotateCW(DRIVE_SPEED);    Serial.print("Rotate CW ");   break;
    default:
      Serial.print("Error: unknown command '");
      Serial.print(dir);
      Serial.println("'");
      return;
  }

  Serial.print(seconds, 2);
  Serial.println(" s");

  moveStartMs = millis();
  moveDurationMs = (unsigned long)(seconds * 1000.0);
  moving = true;
}

void readSerial() {
  while (Serial.available() > 0) {
    char c = Serial.read();
    if (c == '\n' || c == '\r') {
      if (inputBuffer.length() > 0) {
        handleCommand(inputBuffer);
        inputBuffer = "";
      }
    } else {
      inputBuffer += c;
    }
  }
}

// ---------- Setup / loop ----------
void setup() {
  Serial.begin(9600);
  delay(1000);

  runStartupTest();

  // Discard anything typed during the blocking test so it isn't run afterward
  while (Serial.available() > 0) Serial.read();
  inputBuffer = "";

  Serial.println("Ready. Commands: W/S/A/D/Q/E + seconds (e.g. W1, Q1.2). X = stop.");
}

void loop() {
  readSerial();

  if (moving && (millis() - moveStartMs >= moveDurationMs)) {
    stopAllMotors();
    moving = false;
    Serial.println("Done");
  }
}
